# AriaDM P0 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** aria2 원격 제어 모바일 앱 v1(P0) — 프로필 관리, 다운로드 목록·제어, URL/magnet/torrent 추가, 완료 로컬 알림.

**Architecture:** 프로필별 `Aria2Connection`이 WebSocket(이벤트 구동) + HTTP 폴링 fallback으로 aria2 JSON-RPC를 처리한다. 서버 상태는 TanStack Query, 앱 상태는 Zustand+MMKV, secret은 expo-secure-store. expo-router 탭 네비게이션, 플랫폼별 테마 분기.

**Tech Stack:** Expo SDK 57 (expo-router v7), React Native 0.83+, TypeScript strict, TanStack Query v5, Zustand v5, react-native-mmkv, expo-secure-store, expo-notifications, expo-task-manager + expo-background-task, jest-expo, Maestro.

**Spec:** `docs/superpowers/specs/2026-08-22-ariadm-design.md`

## Global Constraints

- 패키지 매니저: `pnpm` (npm/yarn 금지)
- 버전 정책: 설치 시점 최신 (`create-expo-app@latest`, `pnpm add ...@latest`). 기대 기준선: Expo SDK 57

- 모든 rpc-secret·토큰은 repo에 커밋 금지 — `.env`(gitignore) 사용, 키 목록만 `.env.example`

```ts
// aria2 응답은 전부 문자열 숫자 — 표시용 변환 유틸
export function formatBytes(n: number): string {
  if (n < 1024) return `${Math.round(n)} B`;
  const units = ['KB', 'MB', 'GB', 'TB'];
  let v = n / 1024;
  let i = 0;
  while (v >= 1024 && i < units.length - 1) { v /= 1024; i++; }
  return `${v >= 100 ? Math.round(v) : v.toFixed(1)} ${units[i]}`;
}

export function formatSpeed(bytesPerSec: number): string {
  return `${formatBytes(bytesPerSec)}/s`;
}

export function formatEta(seconds: number): string {
  if (!isFinite(seconds) || seconds <= 0) return '-';
  const s = Math.floor(seconds);
  const h = Math.floor(s / 3600);
  const m = Math.floor((s % 3600) / 60);
  const sec = s % 60;
  if (h > 0) return `${h}h ${m}m`;
  if (m > 0) return `${m}m ${sec}s`;
  return `${sec}s`;
}

export function progressPercent(completed: number, total: number): number {
  if (total <= 0) return 0;
  return Math.min(100, Math.floor((completed / total) * 100));
}

// 표시명 우선순위: 토렌트 이름 > 파일 경로 basename > gid
export function downloadName(info: {
  gid: string;
  files?: { path: string }[];
  bittorrent?: { info?: { name?: string } } | null;
}): string {
  const bt = info.bittorrent?.info?.name;
  if (bt && bt.length > 0) return bt;
  const p = info.files?.[0]?.path;
  if (p) {
    const base = p.split('/').filter(Boolean).pop();
    if (base) return base;
  }
  return info.gid;
}
```

`shared/base64.ts`:

```ts
const CHARS = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/';

// Hermes에는 btoa가 없어 자체 인코딩 — torrent 파일 업로드용
export function bytesToBase64(bytes: Uint8Array): string {
  let out = '';
  for (let i = 0; i < bytes.length; i += 3) {
    const b0 = bytes[i];
    const b1 = bytes[i + 1];
    const b2 = bytes[i + 2];
    out += CHARS[b0 >> 2];
    out += CHARS[((b0 & 3) << 4) | ((b1 ?? 0) >> 4)];
    if (b1 === undefined) { out += '=='; break; }
    out += CHARS[((b1 & 15) << 2) | ((b2 ?? 0) >> 6)];
    if (b2 === undefined) { out += '='; break; }
    out += CHARS[b2 & 63];
  }
  return out;
}
```

`shared/i18n/index.ts`:

```ts
import * as Localization from 'expo-localization';
import { en, type DictKey } from './en';
import { ko } from './ko';

// 신규 언어 추가: <code>.ts 사전 생성 → DICTS·availableLangs에 등록
const DICTS: Record<string, Partial<Record<DictKey, string>>> = { en, ko };

export const availableLangs = [
  { code: 'auto', label: 'Auto' },
  { code: 'en', label: 'English' },
  { code: 'ko', label: '한국어' },
];

let effective = 'en';

function resolve(code: string): string {
  return code in DICTS ? code : 'en';
}

// pref 'auto'면 기기 언어 따름 — 앱 시작 시 settings store가 호출
export function initLang(pref: string): void {
  effective = pref === 'auto'
    ? resolve(Localization.getLocales()[0]?.languageCode ?? 'en')
    : resolve(pref);
}

export function setLang(code: string): void { effective = resolve(code); }
export function getLang(): string { return effective; }

export function t(key: string, vars?: Record<string, string | number>): string {
```

```ts
export class RpcError extends Error {
  constructor(public code: number, message: string) {
    super(message);
  }
}

export function buildRequest(id: number, method: string, params: unknown[] = []): string {
  return JSON.stringify({ jsonrpc: '2.0', id, method, params });
}

export type RpcMessage =
  | { kind: 'response'; id: number; result: unknown; error: { code: number; message: string } | null }
  | { kind: 'notification'; method: string; params: unknown[] }
  | { kind: 'invalid' };

export function parseMessage(raw: string): RpcMessage {
  let v: unknown;
  try {
    v = JSON.parse(raw);
  } catch {
    return { kind: 'invalid' };
  }
  if (typeof v !== 'object' || v === null) return { kind: 'invalid' };
  const o = v as Record<string, unknown>;
  if ('method' in o && typeof o.method === 'string' && !('id' in o)) {
    return { kind: 'notification', method: o.method, params: Array.isArray(o.params) ? o.params : [] };
  }
  if ('id' in o && (typeof o.id === 'number' || typeof o.id === 'string')) {
    const err = o.error as { code: number; message: string } | undefined;
    return { kind: 'response', id: Number(o.id), result: o.result, error: err ?? null };
  }
  return { kind: 'invalid' };
}
```

```ts
import { parseMessage, RpcError, buildRequest } from './jsonrpc';
import type { Aria2Event, Aria2EventName, ConnState } from './types';

export interface WSLike {
  send(data: string): void;
  close(): void;
  onopen: (() => void) | null;
  onmessage: ((ev: { data: unknown }) => void) | null;
  onclose: (() => void) | null;
  onerror: ((ev: unknown) => void) | null;
}

export interface ConnectionConfig {
  url: string; // http(s)://host:port
  secret: string;
}

export interface ConnectionOptions {
  httpOnly?: boolean;
  wsFactory?: (url: string) => WSLike;
  fetchImpl?: typeof fetch;
  wsMaxFailures?: number; // 기본 3 — 연속 실패 시 degraded 전환
  degradeRetryMs?: number; // 기본 60_000 — degraded에서 WS 재시도 주기
  callTimeoutMs?: number;  // 기본 8000
}

const NOTIFICATIONS: Record<string, Aria2EventName> = {
  'aria2.onDownloadStart': 'onDownloadStart',
  'aria2.onDownloadComplete': 'onDownloadComplete',
  'aria2.onDownloadError': 'onDownloadError',
  'aria2.onBtDownloadComplete': 'onBtDownloadComplete',
};

export class Aria2Connection {
  private ws: WSLike | null = null;
  private state: ConnState = 'disconnected';
  private nextId = 1;
  private pending = new Map<number, { resolve: (v: unknown) => void; reject: (e: Error) => void; timer: ReturnType<typeof setTimeout> }>();
  private notifCbs = new Set<(e: Aria2Event) => void>();
  private stateCbs = new Set<(s: ConnState) => void>();
  private failureCount = 0;
  private wsFailures = 0;
  private stopped = true;
  private degradeTimer: ReturnType<typeof setTimeout> | null = null;
  private readonly o: Required<Pick<ConnectionOptions, 'wsMaxFailures' | 'degradeRetryMs' | 'backoffBaseMs' | 'callTimeoutMs'>>;

  constructor(
    private config: ConnectionConfig,
    options: ConnectionOptions = {},
  ) {
    this.o = {
      wsMaxFailures: options.wsMaxFailures ?? 3,
      degradeRetryMs: options.degradeRetryMs ?? 60_000,
      backoffBaseMs: options.backoffBaseMs ?? 1000, // 테스트에서 1로 주입해 백오프 압축
      callTimeoutMs: options.callTimeoutMs ?? 8000,
    };
    this.httpOnly = options.httpOnly ?? false;
    this.wsFactory = options.wsFactory ?? ((url) => new WebSocket(url) as unknown as WSLike);
    this.fetchImpl = options.fetchImpl ?? ((u, i) => fetch(u, i));
  }

  private httpOnly: boolean;
  private wsFactory: (url: string) => WSLike;
  private fetchImpl: typeof fetch;

  private get wsUrl(): string {
    const u = this.config.url.replace(/^http/, 'ws'); // http→ws, https→wss
    return `${u}/jsonrpc`;
  }

  private tokenParams(params: unknown[] = []): unknown[] {
    return [`token:${this.config.secret}`, ...params];
  }

  private setState(s: ConnState): void {
    if (this.state === s) return;
    this.state = s;
    for (const cb of this.stateCbs) cb(s);
  }

  start(): void {
    this.stopped = false;
    if (this.httpOnly) { this.setState('degraded'); return; }
    this.connectWs();
  }

  stop(): void {
    this.stopped = true;
    if (this.degradeTimer) clearTimeout(this.degradeTimer);
    this.ws?.close();
    this.ws = null;
    for (const [, p] of this.pending) { clearTimeout(p.timer); p.reject(new Error('connection stopped')); }
    this.pending.clear();
    this.setState('disconnected');
  }

  getState(): ConnState { return this.state; }

  private connectWs(): void {
    if (this.stopped) return;
    this.setState('connecting');
    const ws = this.wsFactory(this.wsUrl);
    this.ws = ws;
    ws.onopen = () => {
      this.wsFailures = 0;
      this.setState('connected');
    };
    ws.onmessage = (ev) => {
      if (typeof ev.data !== 'string') return;
      const m = parseMessage(ev.data);
      if (m.kind === 'notification') {
        const name = NOTIFICATIONS[m.method];
        if (!name) return;
        const gid = (m.params[0] as { gid?: string })?.gid ?? '';
        for (const cb of this.notifCbs) cb({ name, gid });
      } else if (m.kind === 'response') {
        const p = this.pending.get(m.id);
        if (!p) return;
        clearTimeout(p.timer);
        this.pending.delete(m.id);
        if (m.error) p.reject(new RpcError(m.error.code, m.error.message));
        else p.resolve(m.result);
      }
    };
    ws.onerror = () => { /* onclose가 뒤따름 — 거기서 처리 */ };
    ws.onclose = () => {
      if (this.ws !== ws) return; // 이미 교체/폐기된 소켓 — 무시
      this.ws = null;
      // 끊김 시 pending WS 요청 즉시 실패 처리 (timeout 대기 없음)
      for (const [, p] of this.pending) { clearTimeout(p.timer); p.reject(new Error('connection lost')); }
      this.pending.clear();
      if (this.stopped) return;
      this.wsFailures++;
      if (this.wsFailures >= this.o.wsMaxFailures) {
        this.setState('degraded');
        this.degradeTimer = setTimeout(() => {
          if (this.state === 'degraded' && !this.stopped) this.connectWs();
        }, this.o.degradeRetryMs);
      } else {
        setTimeout(() => this.connectWs(), this.o.backoffBaseMs * this.wsFailures); // backoff
      }
    };
  }

  // 백오프/실패 카운트 무시하고 즉시 재시도 — NetInfo 네트워크 복구 트리거용
  // close 순서 주의: 먼저 this.ws를 null로 만들고 close해야 onclose 가드가
  // 동기 close(FakeWs 등)에서도 백오프 재예약을 발동시키지 않는다
  reconnectNow(): void {
    if (this.stopped) return;
    if (this.degradeTimer) clearTimeout(this.degradeTimer);
    this.wsFailures = 0;
    const old = this.ws;
    this.ws = null;
    old?.close();
    if (this.httpOnly) return;
    this.connectWs();
  }

  call<T = unknown>(method: string, params?: unknown[]): Promise<T> {
    if (this.stopped) return Promise.reject(new Error('connection stopped'));
    if (this.ws && this.state === 'connected') return this.callWs<T>(method, params);
    return this.callHttp<T>(method, params);
  }

  private callWs<T>(method: string, params?: unknown[]): Promise<T> {
    const id = this.nextId++;
    const body = buildRequest(id, method, this.tokenParams(params));
    return new Promise<T>((resolve, reject) => {
      const timer = setTimeout(() => {
        this.pending.delete(id);
        reject(new Error(`rpc timeout: ${method}`));
      }, this.o.callTimeoutMs);
      this.pending.set(id, { resolve: resolve as (v: unknown) => void, reject, timer });
      this.ws?.send(body);
    });
  }

  private async callHttp<T>(method: string, params?: unknown[]): Promise<T> {
    const id = this.nextId++;
    const body = buildRequest(id, method, this.tokenParams(params));
    // fetch 자체 timeout 없음 — callTimeoutMs로 경쟁시켜 hang 방지
    const res = await Promise.race([
      this.fetchImpl(`${this.config.url}/jsonrpc`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body,
      }),
      new Promise<never>((_, reject) =>
        setTimeout(() => reject(new Error(`rpc timeout: ${method}`)), this.o.callTimeoutMs)),
    ]);
    const m = parseMessage(await res.text());
    if (m.kind !== 'response') throw new Error('invalid rpc response');
    if (m.error) throw new RpcError(m.error.code, m.error.message);
    return m.result as T;
  }

  onNotification(cb: (e: Aria2Event) => void): () => void {
    this.notifCbs.add(cb);
    return () => this.notifCbs.delete(cb);
  }

  onStateChange(cb: (s: ConnState) => void): () => void {
    this.stateCbs.add(cb);
    return () => this.stateCbs.delete(cb);
  }
}
```

```ts
import { create } from 'zustand';

export interface Profile {
  id: string;
  name: string;
  scheme: 'http' | 'https';
  host: string;
  port: number;
  secret: string;
}

// 비동기 필수 — SecureStore(Keychain/Keystore)가 동기 API를 제공하지 않음
export interface ProfileStorage {
  load(): Promise<{ profiles: Profile[]; activeId: string | null }>;
  save(state: { profiles: Profile[]; activeId: string | null }): Promise<void>;
  deleteSecrets(ids: string[]): Promise<void>; // 프로필 삭제 시 Keychain 잔여 자격 정리
}

interface ProfilesState {
  profiles: Profile[];
  activeId: string | null;
  hydrated: boolean;
  storage: ProfileStorage | null;
  hydrate(storage: ProfileStorage): Promise<void>;
  persist(profiles: Profile[], activeId: string | null): void;
  addProfile(p: Profile): void;
  updateProfile(id: string, patch: Partial<Profile>): void;
  removeProfile(id: string): void;
  setActive(id: string): void;
}

export const useProfiles = create<ProfilesState>((set, get) => ({
  profiles: [],
  activeId: null,
  hydrated: false,
  storage: null,
  async hydrate(storage) {
    const { profiles, activeId } = await storage.load();
    set({ profiles, activeId, hydrated: true, storage });
  },
  persist(profiles: Profile[], activeId: string | null) {
    // fire-and-forget — 저장 실패는 경고만 (UI 차단 안 함)
    get().storage?.save({ profiles, activeId }).catch((e) => console.warn('profile save failed', e));
  },
  addProfile(p) {
    const { profiles, activeId } = get();
    const nextProfiles = [...profiles, p];
    const nextActive = activeId ?? p.id;
    get().persist(nextProfiles, nextActive);
    set({ profiles: nextProfiles, activeId: nextActive });
  },
  updateProfile(id, patch) {
    const { profiles, activeId } = get();
    const nextProfiles = profiles.map((p) => (p.id === id ? { ...p, ...patch } : p));
    get().persist(nextProfiles, activeId);
    set({ profiles: nextProfiles });
  },
  removeProfile(id) {
    const { profiles, activeId, storage } = get();
    const nextProfiles = profiles.filter((p) => p.id !== id);
    const nextActive = activeId === id ? (nextProfiles[0]?.id ?? null) : activeId;
    get().persist(nextProfiles, nextActive);
    // Keychain/Keystore의 secret도 정리 — 잔여 자격 방지
    storage?.deleteSecrets([id]).catch((e) => console.warn('secret delete failed', e));
    set({ profiles: nextProfiles, activeId: nextActive });
  },
  setActive(id) {
    const { profiles } = get();
    get().persist(profiles, id);
    set({ activeId: id });
  },
}));

export function getActiveProfile(state: ProfilesState): Profile | null {
  return state.profiles.find((p) => p.id === state.activeId) ?? null;
}
```

```ts
import { MMKV } from 'react-native-mmkv';
import * as SecureStore from 'expo-secure-store';
import type { Profile, ProfileStorage } from './store';

const mmkv = new MMKV({ id: 'ariadm-profiles' });

// 프로필별 개별 key — 단일 JSON 항목의 크기 제한·부분 실패 회피
const secretKey = (id: string) => `profile-secret-${id}`;

// secret은 SecureStore(getItemAsync/setItemAsync — 동기 API 없음), 나머지는 MMKV.
// 저장 순서: secret 먼저 → metadata 나중 (부분 실패 시 고아 metadata 최소화)
export function createProfileStorage(): ProfileStorage {
  return {
    async load() {
      const raw = mmkv.getString('state');
      if (!raw) return { profiles: [], activeId: null };
      const parsed = JSON.parse(raw) as { profiles: Omit<Profile, 'secret'>[]; activeId: string | null };
      const profiles: Profile[] = [];
      for (const p of parsed.profiles) {
        const secret = await SecureStore.getItemAsync(secretKey(p.id)).catch(() => undefined);
        profiles.push({ ...p, secret: secret ?? '' });
      }
      return { profiles, activeId: parsed.activeId };
    },
    async save({ profiles, activeId }) {
      const lean: Omit<Profile, 'secret'>[] = [];
      for (const { secret, ...rest } of profiles) {
        if (secret) await SecureStore.setItemAsync(secretKey(rest.id), secret);
        lean.push(rest);
      }
      mmkv.set('state', JSON.stringify({ profiles: lean, activeId }));
    },
    async deleteSecrets(ids) {
      for (const id of ids) {
        await SecureStore.deleteItemAsync(secretKey(id)).catch(() => undefined);
      }
    },
  };
}
```

```tsx
import { createContext, useContext, useEffect, useMemo, useRef, useState } from 'react';
import { useQueryClient } from '@tanstack/react-query';
import NetInfo from '@react-native-community/netinfo';
import { Aria2Connection } from './connection';
import type { ConnState } from './types';
import { useProfiles, getActiveProfile } from '../../features/profiles/store';
import { createProfileStorage } from '../../features/profiles/storage';

interface Ctx {
  conn: Aria2Connection | null;
  state: ConnState;
  profile: ReturnType<typeof getActiveProfile>;
}

const ConnectionCtx = createContext<Ctx>({ conn: null, state: 'disconnected', profile: null });

export function ConnectionProvider({ children }: { children: React.ReactNode }) {
  const storage = useMemo(() => createProfileStorage(), []);
  const hydrated = useProfiles((s) => s.hydrated);
  const [, force] = useState(0);
  useEffect(() => {
    if (!hydrated) useProfiles.getState().hydrate(storage).then(() => force((n) => n + 1));
  }, [hydrated, storage]);

  const profile = useProfiles(getActiveProfile);
  const [conn, setConn] = useState<Aria2Connection | null>(null);
  const [state, setState] = useState<ConnState>('disconnected');
  const qc = useQueryClient();
  const connRef = useRef<Aria2Connection | null>(null);

  // 연결 key — id뿐 아니라 접속 정보·secret 변경 시에도 재연결
  const connKey = profile
    ? `${profile.id}|${profile.scheme}://${profile.host}:${profile.port}|${profile.secret}`
    : '';

  useEffect(() => {
    connRef.current?.stop();
    if (!profile) { setConn(null); setState('disconnected'); return; }
    const c = new Aria2Connection({
      url: `${profile.scheme}://${profile.host}:${profile.port}`,
      secret: profile.secret,
    });
    connRef.current = c;
    setConn(c);
    const offState = c.onStateChange(setState);
    // aria2 이벤트 → 캐시 무효화로 실시간 갱신
    const offNotif = c.onNotification(() => {
      qc.invalidateQueries({ queryKey: ['downloads'] });
      qc.invalidateQueries({ queryKey: ['globalStat'] });
    });
    c.start();
    // NetInfo: 네트워크 복구 시 백오프 무시 즉시 재연결
    const netSub = NetInfo.addEventListener((n) => {
      if (n.isConnected) c.reconnectNow();
    });
    return () => { offState(); offNotif(); netSub(); c.stop(); };
  }, [connKey, qc]);

  return <ConnectionCtx.Provider value={{ conn, state, profile }}>{children}</ConnectionCtx.Provider>;
}

export function useConnection(): Ctx {
  return useContext(ConnectionCtx);
}
```

```ts
import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query';
import { Aria2 } from '../../core/rpc/methods';
import { useConnection } from '../../core/rpc/ConnectionContext';

export type ListKind = 'active' | 'waiting' | 'stopped';

// 폴링 주기 — WS 연결이면 이벤트가 갱신을 담당하므로 여유 있게, degraded면 빠르게
export function useDownloads(kind: ListKind, limit = 100) {
  const { conn, state } = useConnection();
  return useQuery({
    queryKey: ['downloads', kind, limit],
    enabled: !!conn,
    refetchInterval: state === 'connected' ? 3000 : 2000,
    queryFn: async () => {
      if (!conn) return [];
      if (kind === 'active') return Aria2.tellActive(conn);
      if (kind === 'waiting') return Aria2.tellWaiting(conn, 0, limit);
      return Aria2.tellStopped(conn, 0, limit);
    },
  });
}

export function useGlobalStat() {
  const { conn } = useConnection();
  return useQuery({
    queryKey: ['globalStat'],
    enabled: !!conn,
    refetchInterval: 1000,
    queryFn: () => (conn ? Aria2.getGlobalStat(conn) : null),
  });
}

export function useControl() {
  const { conn } = useConnection();
  const qc = useQueryClient();
  const inv = () => qc.invalidateQueries({ queryKey: ['downloads'] });
  return {
    pause: useMutation({ mutationFn: (gid: string) => Aria2.pause(conn!, gid), onSuccess: inv }),
    resume: useMutation({ mutationFn: (gid: string) => Aria2.unpause(conn!, gid), onSuccess: inv }),
    remove: useMutation({ mutationFn: (gid: string) => Aria2.remove(conn!, gid), onSuccess: inv }),
    deleteRecord: useMutation({ mutationFn: (gid: string) => Aria2.removeDownloadResult(conn!, gid), onSuccess: inv }),
  };
}
```

```tsx
import { useState } from 'react';
import { ActivityIndicator, Alert, Text, TextInput, TouchableOpacity, View } from 'react-native';
import * as DocumentPicker from 'expo-document-picker';
import * as LegacyFS from 'expo-file-system/legacy';
import { useRouter } from 'expo-router';
import { Aria2, type AddOptions } from '../../core/rpc/methods';
import { useConnection } from '../../core/rpc/ConnectionContext';
import { bytesToBase64 } from '../../shared/base64';
import { t } from '../../shared/i18n';
import { theme } from '../../shared/theme';

export function AddForm() {
  const { conn, profile } = useConnection();
  const [uri, setUri] = useState('');
  const [dir, setDir] = useState('');
  const [out, setOut] = useState('');
  const [split, setSplit] = useState('');
  const [busy, setBusy] = useState(false);
  const router = useRouter();

  // aria2 option 값은 전부 string — 여기서 한 곳에서 변환
  function buildOptions(): AddOptions | undefined {
    const o: AddOptions = {};
    if (dir.trim()) o.dir = dir.trim();
    if (out.trim()) o.out = out.trim();
    if (split.trim() && Number(split) > 0) o.split = String(Math.floor(Number(split)));
    return Object.keys(o).length ? o : undefined;
  }

  async function submitUri() {
    const u = uri.trim();
    if (!conn || busy || !(u.startsWith('http://') || u.startsWith('https://') || u.startsWith('magnet:'))) return;
    setBusy(true);
    try {
      await Aria2.addUri(conn, [u], buildOptions());
      Alert.alert('AriaDM', t('added', { n: 1 }));
      setUri('');
      router.push('/');
    } catch (e) {
      Alert.alert('AriaDM', `${t('add_failed')}: ${(e as Error).message}`);
    } finally {
      setBusy(false);
    }
  }

  async function pickTorrent() {
    if (!conn || busy) return;
    // '*/*' — iOS UTI 변환·Android octet-stream으로 .torrent가 걸러지는 문제 회피, 확장자로 직접 검증
    const picked = await DocumentPicker.getDocumentAsync({ type: '*/*' });
    if (picked.canceled || !picked.assets[0]) return;
    if (!picked.assets[0].name?.toLowerCase().endsWith('.torrent')) {
      Alert.alert('AriaDM', t('add_failed'));
      return;
    }
    setBusy(true);
    try {
      // 1차: legacy API base64 read — Android content:// URI 처리 겸 네이티브 인코딩
      let b64: string;
      try {
        b64 = await LegacyFS.readAsStringAsync(picked.assets[0].uri, {
          encoding: LegacyFS.EncodingType.Base64,
        });
      } catch {
        // 2차: legacy 미지원 시 new File API + 자체 base64 인코딩
        const f = new (await import('expo-file-system')).File(picked.assets[0].uri);
        b64 = bytesToBase64(await f.bytes());
      }
      await Aria2.addTorrent(conn, b64, buildOptions());
      Alert.alert('AriaDM', t('added', { n: 1 }));
      router.push('/');
    } catch (e) {
      Alert.alert('AriaDM', `${t('add_failed')}: ${(e as Error).message}`);
    } finally {
      setBusy(false);
    }
  }

  if (!profile) return <Text style={{ color: theme.muted, textAlign: 'center', marginTop: 40 }}>{t('empty_no_profile')}</Text>;

  return (
    <View style={{ flex: 1, backgroundColor: theme.bg, padding: 16, gap: 12 }}>
      <TextInput
        value={uri}
        onChangeText={setUri}
        placeholder={t('add_uri_placeholder')}
        autoCapitalize="none"
        autoCorrect={false}
        keyboardType="url"
        style={{ backgroundColor: theme.card, borderRadius: theme.radius, padding: 14, color: theme.text }}
      />
      <TextInput
        value={dir}
        onChangeText={setDir}
        placeholder={t('add_dir_placeholder')}
        autoCapitalize="none"
        autoCorrect={false}
        style={{ backgroundColor: theme.card, borderRadius: theme.radius, padding: 14, color: theme.text }}
      />
      <View style={{ flexDirection: 'row', gap: 12 }}>
        <TextInput
          value={out}
          onChangeText={setOut}
          placeholder={t('add_out_placeholder')}
          autoCapitalize="none"
          autoCorrect={false}
          style={{ flex: 1, backgroundColor: theme.card, borderRadius: theme.radius, padding: 14, color: theme.text }}
        />
        <TextInput
          value={split}
          onChangeText={setSplit}
          placeholder={t('add_split_placeholder')}
          keyboardType="number-pad"
          style={{ flex: 1, backgroundColor: theme.card, borderRadius: theme.radius, padding: 14, color: theme.text }}
        />
      </View>
      <TouchableOpacity onPress={submitUri} disabled={busy}
        style={{ backgroundColor: theme.accent, borderRadius: theme.radius, padding: 16, alignItems: 'center', opacity: busy ? 0.5 : 1 }}>
        {busy ? <ActivityIndicator color="#fff" /> : <Text style={{ color: '#fff', fontWeight: '600' }}>{t('add_submit')}</Text>}

        <Text style={{ color: theme.accent }}>{t('add_pick_torrent')}</Text>
      </TouchableOpacity>
    </View>
  );
}
```

```ts
export type DiagResult = { ok: true; version: string } | { ok: false; kind: 'auth' | 'unreachable' };

// RN fetch는 포트 차단/TLS/네트워크 오류를 구분하지 않음 — unreachable로 통합
export async function diagnose(
  cfg: { scheme: 'http' | 'https'; host: string; port: number; secret: string },
  fetchImpl: typeof fetch = ((u: string, i?: RequestInit) => fetch(u, i)) as typeof fetch,
): Promise<DiagResult> {
  try {
    const res = await fetchImpl(`${cfg.scheme}://${cfg.host}:${cfg.port}/jsonrpc`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        jsonrpc: '2.0', id: 1, method: 'aria2.getVersion',
        params: [`token:${cfg.secret}`],
      }),
    });
    const body = JSON.parse(await res.text()) as {
      result?: { version?: string };
      error?: { code: number };
    };
    if (body.error) return { ok: false, kind: 'auth' };
    return { ok: true, version: body.result?.version ?? '?' };
  } catch {
    return { ok: false, kind: 'unreachable' };
  }
}
```

```ts
import { Aria2 } from '../rpc/methods';
import { computeNewCompletions } from './diff';
import { downloadName } from '../../shared/format';

export interface SeenStore {
  load(): string[];
  save(gids: string[]): void;
}

// background task와 단위 테스트가 공유하는 핵심 로직.
// 첫 실행(seen 비어 있음)은 baseline 초기화 — 기존 완료 전부를 알림 없이 seen으로 등록 (알림 폭주 방지).
// 한계: tellStopped(0, 200) 창은 최근 200건 — 200 초과 신규 완료는 다음 주기에 걸릴 수 있음 (seen 상한 300과 정합)
export async function runCompletionCheck(
  conn: Parameters<typeof Aria2.tellStopped>[0],
  seenStore: SeenStore,
  notify: (name: string) => Promise<void>,
): Promise<number> {
  const stopped = await Aria2.tellStopped(conn, 0, 200);
  const loaded = seenStore.load();
  const seen = new Set(loaded);
  if (loaded.length === 0) {
    // baseline — 현재 완료 전부 기록, 알림 없음
    seenStore.save(stopped.filter((d) => d.status === 'complete').map((d) => d.gid));
    return 0;
  }
  const fresh = computeNewCompletions(seen, stopped);
  for (const gid of fresh) {
    const info = await Aria2.tellStatus(conn, gid).catch(() => null);
    await notify(info ? downloadName(info) : gid);
    seen.add(gid);
  }
  seenStore.save([...seen].slice(-300));
  return fresh.length;
}
```

```ts
import * as TaskManager from 'expo-task-manager';
import * as BackgroundTask from 'expo-background-task';
import { MMKV } from 'react-native-mmkv';
import { Aria2Connection } from '../rpc/connection';
import { createProfileStorage } from '../../features/profiles/storage';
import { useProfiles, getActiveProfile } from '../../features/profiles/store';
import { runCompletionCheck, type SeenStore } from './check';
import { notifyComplete } from './notifications';

export const BG_TASK = 'ariadm-completion-check';

const seenMmkv = new MMKV({ id: 'ariadm-seen' });
export const mmkvSeen: SeenStore = {
  load: () => JSON.parse(seenMmkv.getString('seen') ?? '[]') as string[],
  save: (gids) => seenMmkv.set('seen', JSON.stringify(gids.slice(-300))),
};

// 모듈 최상위 정의 — background headless 기동 시 React 컴포넌트가 마운트되지 않아도
// JS 번들 로드만으로 task가 등록된다. 프로필은 실행 시점에 저장소에서 조회 (closure 캡처 금지).
TaskManager.defineTask(BG_TASK, async () => {
  const profilesState = useProfiles.getState();
  if (!profilesState.hydrated) await profilesState.hydrate(createProfileStorage()); // headless 기동 대비
  const profile = getActiveProfile(useProfiles.getState());
  if (!profile) return;
  const conn = new Aria2Connection(
    { url: `${profile.scheme}://${profile.host}:${profile.port}`, secret: profile.secret },
    { httpOnly: true },
  );
  conn.start();
  try {
    await runCompletionCheck(conn, mmkvSeen, notifyComplete);
  } finally {
    conn.stop();
  }
});

export async function registerBackgroundTask(): Promise<void> {
  if (await TaskManager.isTaskRegisteredAsync(BG_TASK)) return;
  // minimumInterval 단위는 '분' — 15 = 15분 (15*60은 15시간이 됨)
  await BackgroundTask.registerTaskAsync(BG_TASK, { minimumInterval: 15 });
}
```

```tsx
import { ensurePermission } from '../../core/notifications/notifications';
import { isPro } from '../../core/iap/entitlement';
import { availableLangs, t } from '../../shared/i18n';
import { theme } from '../../shared/theme';

export default function SettingsScreen() {
  const router = useRouter();
  const { profiles, activeId, setActive } = useProfiles();
  const { notifyOnComplete, setNotify, lang, setLang } = useSettings();

  return (
    <ScrollView style={{ flex: 1, backgroundColor: theme.bg, padding: 16, gap: 16 }}>
      <Text style={{ fontWeight: '600', color: theme.text }}>{t('settings_profiles')}</Text>
      {profiles.map((p) => (
        <Pressable key={p.id} onPress={() => setActive(p.id)}
          onLongPress={() => Alert.alert(p.name, '', [
            { text: t('profile_edit'), onPress: () => router.push(`/modal/profile?id=${p.id}`) },
            { text: t('profile_delete'), style: 'destructive' as const,
              onPress: () => useProfiles.getState().removeProfile(p.id) },
            { text: 'Cancel', style: 'cancel' as const },
          ])}
          style={{ backgroundColor: theme.card, borderRadius: theme.radius, padding: 14 }}>
          <Text style={{ color: p.id === activeId ? theme.accent : theme.text, fontWeight: p.id === activeId ? '700' : '400' }}>{p.name}</Text>
          <Text style={{ color: theme.muted, fontSize: 12 }}>{p.scheme}://{p.host}:{p.port}</Text>
        </Pressable>
      ))}
      <Pressable onPress={() => router.push('/modal/profile')}
        style={{ padding: 14, borderRadius: theme.radius, borderWidth: 1, borderColor: theme.accent, alignItems: 'center' }}>
        <Text style={{ color: theme.accent }}>+ {t('profile_title')}</Text>
      </Pressable>

      <View style={{ flexDirection: 'row', alignItems: 'center', gap: 8 }}>
        <Switch value={notifyOnComplete} onValueChange={async (v) => { if (v) await ensurePermission(); setNotify(v); }} />
        <View style={{ flex: 1 }}>
          <Text style={{ color: theme.text }}>{t('settings_notify')}</Text>
          <Text style={{ color: theme.muted, fontSize: 12 }}>{t('settings_notify_note')}</Text>
        </View>
      </View>

      <Text style={{ fontWeight: '600', color: theme.text }}>{t('settings_lang')}</Text>
      <View style={{ flexDirection: 'row', gap: 8, flexWrap: 'wrap' }}>
        {availableLangs.map((l) => (
          <Pressable key={l.code} onPress={() => setLang(l.code)}
            style={{ padding: 10, borderRadius: theme.radius, backgroundColor: lang === l.code ? theme.accent : theme.card }}>
            <Text style={{ color: lang === l.code ? '#fff' : theme.text }}>
              {l.code === 'auto' ? t('settings_lang_auto') : l.label}
            </Text>
          </Pressable>
        ))}
      </View>

      <Text style={{ color: theme.muted, fontSize: 12 }}>v1 — Free (Pro: {String(isPro())})</Text>
    </ScrollView>
  );
}
```
